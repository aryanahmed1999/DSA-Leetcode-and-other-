class Solution {
public:
    int removeDuplicates(vector<int>& nums) {
        int size = nums.size();
        int index = 0 ;
        int elements = 1;
        while(index < size - 1){
            if(nums[index] != nums[index + 1]){
                nums[elements] = nums[index + 1];
                elements++;
            }
            index ++;
        }
        return elements;
    }
};
